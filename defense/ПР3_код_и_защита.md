# ПР3 — код и объяснение для защиты

## Суть работы

`fpga_top` содержит 64 подготовленных входа и 64 эталонных бита. Автомат подаёт входы на ядро, сверяет ответы и ведёт счётчики `checked_count` и `error_count`; после последнего вектора устанавливает `done` и `pass`. Тестбенч проверяет правильный прогон, обнаружение намеренной ошибки в эталоне и повторный правильный прогон. Фиксированные 64 входа нужны для самопроверки прототипа, но не подходят для честного сравнения размеров и частоты ядра: синтезатор может упростить вычисления под известные заранее значения.

Для сравнения `bnn_benchmark_baseline` и `bnn_benchmark_pipelined` получают переменное 40-битное слово через одинаковый входной регистр. Оба классификатора используют одни и те же модули слоёв A и B; единственное архитектурное отличие — регистр между ними. Между ними регистрируются восемь бит скрытого слоя и `valid`. Оба варианта способны принять новый вход каждый такт; конвейер увеличивает задержку отдельного ответа на один такт. Поэтому ускорение оценивается по периодам и WNS после трассировки при одинаковых условиях, а расход — по LUT и регистрам. Число тактов старого автомата подачи не считается выигрышем от регистра между слоями.

Для защиты открой единый `vivado_pr23/bnn_pr23.xpr`. Покажи исходники `bnn_layer_a.v`, `bnn_layer_b.v`, `bnn_classifier_pipelined.v`, симуляции `sim_pr3` и `sim_pr3_pipeline`, затем отчёты `baseline_20`, `pipeline_20`, а также результаты 10 нс. Демонстрируй размещение в Device и проложенные цепи по `route_status.rpt`. При отсутствии платы не делай вывод о работе реального I/O: распиновка и внешние задержки не заданы, битовый поток не создавался.

### `rtl/bnn_layer_a.v`

```verilog
`timescale 1ns / 1ps

// Five signed int8 features -> eight thresholded hidden neurons.
module bnn_layer_a #(
    parameter W_FILE = "weights_layerA_W.mem",
    parameter B_FILE = "weights_layerA_b.mem"
) (
    input  wire [39:0] input_features,
    output reg  [7:0]  hidden_bits
);
    reg [7:0]  weights [0:39];
    reg [31:0] biases [0:7];

    initial begin
        $readmemh(W_FILE, weights);
        $readmemh(B_FILE, biases);
    end

    reg signed [15:0] product;
    reg signed [31:0] accumulator;
    integer h;
    integer j;

    always @* begin
        product = 16'sd0;
        accumulator = 32'sd0;
        hidden_bits = 8'b0;
        for (h = 0; h < 8; h = h + 1) begin
            accumulator = $signed(biases[h]);
            for (j = 0; j < 5; j = j + 1) begin
                product = $signed(input_features[39 - 8*j -: 8])
                        * $signed(weights[j*8 + h]);
                accumulator = accumulator + {{16{product[15]}}, product};
            end
            hidden_bits[h] = (accumulator >= 32'sd0);
        end
    end
endmodule

```

### `rtl/bnn_layer_b.v`

```verilog
`timescale 1ns / 1ps

// Eight binary activations -> one binary decision.
module bnn_layer_b #(
    parameter W_FILE = "weights_layerB_W.mem",
    parameter B_FILE = "weights_layerB_b.mem"
) (
    input  wire [7:0]         hidden_bits,
    output reg  signed [31:0] score,
    output reg                prediction
);
    reg [0:0]  weights [0:7];
    reg [31:0] biases [0:0];

    initial begin
        $readmemh(W_FILE, weights);
        $readmemh(B_FILE, biases);
    end

    reg [3:0] matching_bits;
    integer h;

    always @* begin
        matching_bits = 4'b0;
        for (h = 0; h < 8; h = h + 1)
            matching_bits = matching_bits + (hidden_bits[h] ~^ weights[h][0]);
        score = ($signed({28'b0, matching_bits}) <<< 1)
              - 32'sd8 + $signed(biases[0]);
        prediction = (score >= 32'sd0);
    end
endmodule

```

```verilog
// Same two layer modules as the pipelined core, without an interlayer register.
module bnn_classifier_modular #(
    parameter W_A_FILE = "weights_layerA_W.mem",
    parameter B_A_FILE = "weights_layerA_b.mem",
    parameter W_B_FILE = "weights_layerB_W.mem",
    parameter B_B_FILE = "weights_layerB_b.mem"
) (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        valid_in,
    input  wire [39:0] input_features,
    output reg         valid_out,
    output reg         fall_detected
);
    wire [7:0] hidden_bits;
    wire signed [31:0] score;
    wire prediction_next;

    bnn_layer_a #(.W_FILE(W_A_FILE), .B_FILE(B_A_FILE)) layer_a (
        .input_features(input_features),
        .hidden_bits(hidden_bits)
    );

    bnn_layer_b #(.W_FILE(W_B_FILE), .B_FILE(B_B_FILE)) layer_b (
        .hidden_bits(hidden_bits),
        .score(score),
        .prediction(prediction_next)
    );

    always @(posedge clk) begin
        if (!rst_n) begin
            valid_out <= 1'b0;
            fall_detected <= 1'b0;
        end else begin
            valid_out <= valid_in;
            if (valid_in)
                fall_detected <= prediction_next;
        end
    end
endmodule
```

```verilog
// One register boundary between layers A and B; accepts a vector every clock.
module bnn_classifier_pipelined #(
    parameter W_A_FILE = "weights_layerA_W.mem",
    parameter B_A_FILE = "weights_layerA_b.mem",
    parameter W_B_FILE = "weights_layerB_W.mem",
    parameter B_B_FILE = "weights_layerB_b.mem"

    input  wire        clk,
    input  wire        rst_n,
    input  wire        valid_in,
    input  wire [39:0] input_features,
    output reg         valid_out,
    output reg         fall_detected
);
    wire [7:0] hidden_next;
    reg [7:0] hidden_q;
    reg valid_hidden_q;
    wire signed [31:0] score;
    wire prediction_next;

    bnn_layer_a #(.W_FILE(W_A_FILE), .B_FILE(B_A_FILE)) layer_a (
        .input_features(input_features),
        .hidden_bits(hidden_next)
    );

    bnn_layer_b #(.W_FILE(W_B_FILE), .B_FILE(B_B_FILE)) layer_b (
        .hidden_bits(hidden_q),
        .score(score),
        .prediction(prediction_next)

        if (!rst_n) begin
            hidden_q <= 8'b0;
            valid_hidden_q <= 1'b0;
            valid_out <= 1'b0;
            fall_detected <= 1'b0;

            hidden_q <= hidden_next;
            valid_hidden_q <= valid_in;
            valid_out <= valid_hidden_q;
            if (valid_hidden_q)
                fall_detected <= prediction_next;
        end
    end
endmodule
```

```verilog
module tb_bnn_classifier_pipelined;

`elsif EDGE_TEST
    localparam VECTOR_COUNT = 14;

    reg [31:0] expected_score [0:VECTOR_COUNT-1];

    reg [31:0] expected_score [0:VECTOR_COUNT-1];
`endif
    integer i;
    integer errors;

    bnn_classifier_pipelined dut (
        .clk(clk),
        .rst_n(rst_n),
        .valid_in(valid_in),
        .input_features(input_features),
        .valid_out(valid_out),
        .fall_detected(fall_detected)
    );

    initial begin
`ifdef FULL_TEST
        $readmemh("tb_inputs_full.mem", vectors);
        $readmemh("tb_expected_full.mem", expected);
        $readmemh("tb_hidden_full.mem", expected_hidden);
        $readmemh("tb_score_full.mem", expected_score);
`elsif EDGE_TEST
        $readmemh("tb_inputs_edge.mem", vectors);
        $readmemh("tb_expected_edge.mem", expected);
        $readmemh("tb_hidden_edge.mem", expected_hidden);
        $readmemh("tb_score_edge.mem", expected_score);
`else
        $readmemh("tb_inputs_64.mem", vectors);
        $readmemh("tb_expected_64.mem", expected);
`endif
        errors = 0;
        repeat (2) @(negedge clk);
        if (valid_out !== 1'b0 || fall_detected !== 1'b0)
            $fatal(1, "reset did not clear outputs");
        rst_n = 1'b1;

        for (i = 0; i < VECTOR_COUNT + 2; i = i + 1) begin
            @(negedge clk);
            valid_in = (i < VECTOR_COUNT);
            input_features = (i < VECTOR_COUNT) ? vectors[i] : 40'b0;
            @(posedge clk);
            #1;
`ifdef FULL_TEST
            if (i < VECTOR_COUNT) begin
                if (dut.hidden_q !== expected_hidden[i]) begin
                    $display("FAIL hidden %0d", i);
                    errors = errors + 1;
                end
                if (dut.score !== $signed(expected_score[i])) begin
                    $display("FAIL score %0d", i);
                    errors = errors + 1;
                end
            end
`elsif EDGE_TEST
            if (i < VECTOR_COUNT) begin
                if (dut.hidden_q !== expected_hidden[i]) begin
                    $display("FAIL hidden %0d", i);
                    errors = errors + 1;
                end
                if (dut.score !== $signed(expected_score[i])) begin
                    $display("FAIL score %0d", i);
                    errors = errors + 1;
                end
            end
`endif
            if (i >= 1 && i - 1 < VECTOR_COUNT) begin
                if (valid_out !== 1'b1 || fall_detected !== expected[i-1]) begin
                    $display("FAIL output %0d: valid=%b got=%b expected=%b",
                             i-1, valid_out, fall_detected, expected[i-1]);
                    errors = errors + 1;
                end
            end else if (valid_out !== 1'b0) begin
                $display("FAIL early valid at cycle %0d", i);
                errors = errors + 1;
            end
        end

        @(negedge clk);
        valid_in = 1'b0;
        @(posedge clk);
        #1;
        if (valid_out !== 1'b0) begin
            $display("FAIL valid_out did not clear");
            errors = errors + 1;
        end
        if (errors != 0)
            $fatal(1, "%0d failures in %0d vectors", errors, VECTOR_COUNT);
        $display("PASS: pipelined classifier, %0d vectors", VECTOR_COUNT);
        $finish;
    end
endmodule
```

### `fpga/fpga_top.v`

```verilog
`timescale 1ns / 1ps

// Board-independent FPGA prototype. A small ROM feeds the classifier with
// 64 reference inputs prepared in practice 1; the outputs are checked on chip.
module fpga_top (
    input  wire       clk,
    input  wire       rst_n,
    input  wire       start,
    output reg        busy,
    output reg        done,
    output reg        pass,
    output reg [6:0]  checked_count,
    output reg [6:0]  error_count,
    output reg        last_prediction
);
    localparam [1:0] IDLE = 2'd0;
    localparam [1:0] DRIVE = 2'd1;
    localparam [1:0] CHECK = 2'd2;

    reg [1:0] state;
    reg [5:0] vector_index;
    reg [39:0] vectors [0:63];
    reg [0:0] expected [0:63];

    initial begin
        $readmemh("tb_inputs_64.mem", vectors);
        $readmemh("tb_expected_64.mem", expected);
    end

    wire [39:0] input_features = vectors[vector_index];
    wire valid_in = (state == DRIVE);
    wire valid_out;
    wire fall_detected;
    wire mismatch = (valid_out != 1'b1) ||
                    (fall_detected != expected[vector_index][0]);

    bnn_classifier classifier (
        .clk(clk),
        .rst_n(rst_n),
        .valid_in(valid_in),
        .input_features(input_features),
        .valid_out(valid_out),
        .fall_detected(fall_detected)
    );

    always @(posedge clk) begin
        if (!rst_n) begin
            state <= IDLE;
            vector_index <= 6'd0;
            busy <= 1'b0;
            done <= 1'b0;
            pass <= 1'b0;
            checked_count <= 7'd0;
            error_count <= 7'd0;
            last_prediction <= 1'b0;
        end else begin
            case (state)
                IDLE: begin
                    if (start) begin
                        vector_index <= 6'd0;
                        busy <= 1'b1;
                        done <= 1'b0;
                        pass <= 1'b0;
                        checked_count <= 7'd0;
                        error_count <= 7'd0;
                        state <= DRIVE;
                    end
                end
                DRIVE: state <= CHECK;
                CHECK: begin
                    checked_count <= checked_count + 7'd1;
                    last_prediction <= fall_detected;
                    if (mismatch)
                        error_count <= error_count + 7'd1;
                    if (vector_index == 6'd63) begin
                        busy <= 1'b0;
                        done <= 1'b1;
                        pass <= (error_count == 7'd0) && !mismatch;
                        state <= IDLE;
                    end else begin
                        vector_index <= vector_index + 6'd1;
                        state <= DRIVE;
                    end
                end
                default: state <= IDLE;
            endcase
        end
    end
endmodule

```

```verilog
// Streaming version of the FPGA self-test for the pipelined classifier.
module fpga_top_pipelined (
    input  wire       clk,
    input  wire       rst_n,
    input  wire       start,
    output reg        busy,
    output reg        done,
    output reg        pass,
    output reg [6:0]  checked_count,
    output reg [6:0]  error_count,
    output reg        last_prediction
);
    localparam [1:0] IDLE = 2'd0;
    localparam [1:0] SEND = 2'd1;
    localparam [1:0] DRAIN = 2'd2;

    reg [1:0] state;
    reg [5:0] send_index;
    reg [5:0] check_index_q0;
    reg [5:0] check_index_q1;
    reg [39:0] vectors [0:63];
    reg [0:0] expected [0:63];

    initial begin
        $readmemh("tb_inputs_64.mem", vectors);
        $readmemh("tb_expected_64.mem", expected);
    end

    wire [39:0] input_features = vectors[send_index];
    wire valid_in = (state == SEND);
    wire valid_out;
    wire fall_detected;
    wire mismatch = (fall_detected != expected[check_index_q1][0]);

    bnn_classifier_pipelined classifier (
        .clk(clk),
        .rst_n(rst_n),
        .valid_in(valid_in),
        .input_features(input_features),
        .valid_out(valid_out),
        .fall_detected(fall_detected)
    );

    always @(posedge clk) begin
        if (!rst_n) begin
            state <= IDLE;
            send_index <= 6'd0;
            check_index_q0 <= 6'd0;
            check_index_q1 <= 6'd0;
            busy <= 1'b0;
            done <= 1'b0;
            pass <= 1'b0;
            checked_count <= 7'd0;
            error_count <= 7'd0;
            last_prediction <= 1'b0;
        end else begin
            check_index_q0 <= send_index;
            check_index_q1 <= check_index_q0;
            case (state)
                IDLE: begin
                    if (start) begin
                        send_index <= 6'd0;
                        busy <= 1'b1;
                        done <= 1'b0;
                        pass <= 1'b0;
                        checked_count <= 7'd0;
                        error_count <= 7'd0;
                        state <= SEND;
                    end
                end
                SEND: begin
                    if (send_index == 6'd63)
                        state <= DRAIN;
                    else
                        send_index <= send_index + 6'd1;
                end
                DRAIN: begin end
                default: state <= IDLE;
            endcase
            if (valid_out && state != IDLE) begin
                checked_count <= checked_count + 7'd1;
                last_prediction <= fall_detected;
                if (mismatch)
                    error_count <= error_count + 7'd1;
                if (checked_count == 7'd63) begin
                    busy <= 1'b0;
                    done <= 1'b1;
                    pass <= (error_count == 7'd0) && !mismatch;
                    state <= IDLE;
                end
            end
        end
    end
endmodule
```

```verilog
// Common registered input for a fair internal clock-path comparison.
// The input features remain variable instead of being fixed in a test ROM.
module bnn_benchmark_shell #(
    parameter PIPELINED = 0
) (
    input  wire        clk,
    input  wire        rst_n,
    input  wire        valid_in,
    input  wire [39:0] input_features,
    output wire        valid_out,
    output wire        fall_detected
);
    reg [39:0] input_q;
    reg valid_q;

    always @(posedge clk) begin
        if (!rst_n) begin
            input_q <= 40'b0;
            valid_q <= 1'b0;
        end else begin
            input_q <= input_features;
            valid_q <= valid_in;
        end
    end

    generate
        if (PIPELINED) begin : with_register
            bnn_classifier_pipelined classifier (
                .clk(clk), .rst_n(rst_n), .valid_in(valid_q),
                .input_features(input_q),
```

```verilog
`timescale 1ns / 1ps

module tb_fpga_top;
    reg clk = 1'b0;
    always #5 clk = ~clk;

    reg rst_n = 1'b0;
    reg start = 1'b0;
    wire busy;
    wire done;
    wire pass;
    wire [6:0] checked_count;
    wire [6:0] error_count;
    wire last_prediction;
    reg saved_expected;
    integer cycles;

    fpga_top dut (
        .clk(clk),
        .rst_n(rst_n),
        .start(start),
        .busy(busy),
        .done(done),
        .pass(pass),
        .checked_count(checked_count),
        .error_count(error_count),
        .last_prediction(last_prediction)
    );

    task run_once;
        input [6:0] wanted_errors;
        begin
            @(negedge clk);
            start = 1'b1;
```
